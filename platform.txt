name=ChipIntelli CI130X Arduino
version=1.0.7
build.ci_service_url=https://gen.yiyu.pro/ci

# CI130X SDK ALG V2.7.14 is ABI-locked to the vendor Nuclei GCC 9.2.0
# distribution. Boards Manager removes the archive's single gcc_fix_raissrc/
# wrapper directory, so the installed tool root contains bin/ directly.
compiler.path={runtime.tools.riscv-gcc.path}/bin/
compiler.prefix=riscv-nuclei-elf-
# The vendor archives were emitted as 32-bit Windows-host GCC LTO and cannot be
# consumed by 64-bit Linux/macOS GCC processes. Those hosts use archives whose
# LTO was materialized once with the validated Windows compiler; Windows keeps
# the original archives and whole-program optimization.
compiler.sdk.library.path={runtime.platform.path}/tools/sdk/lib
compiler.sdk.library.path.linux={runtime.platform.path}/tools/sdk/lib-cross-host
compiler.sdk.library.path.macosx={runtime.platform.path}/tools/sdk/lib-cross-host

compiler.c.cmd={compiler.prefix}gcc
compiler.cpp.cmd={compiler.prefix}g++
compiler.S.cmd={compiler.prefix}gcc
compiler.c.elf.cmd={compiler.prefix}g++
compiler.ar.cmd={compiler.prefix}gcc-ar
compiler.objcopy.cmd={compiler.prefix}objcopy
compiler.size.cmd={compiler.prefix}size

compiler.common.flags=-march=rv32imafc -mabi=ilp32f -mcmodel=medlow -msmall-data-limit=8 -msave-restore -mfdiv -Os -flto -fsigned-char -ffunction-sections -fdata-sections -fno-common -fno-delete-null-pointer-checks -fno-unroll-loops -fshort-enums
compiler.sdk.defines=-DASR_CODE_VERSION=2 -DCORE_ID=0
compiler.sdk.includes="-I{build.variant.path}" "-I{runtime.platform.path}/tools/sdk/include" "-I{runtime.platform.path}/tools/sdk/include/sdk"

compiler.c.flags={compiler.common.flags} -g -std=gnu11 -MMD
# GCC 9.2 may fail to normalize its relative multilib C++ include path when
# Boards Manager is installed below a deep Windows data directory.
compiler.cpp.flags={compiler.common.flags} -g -std=gnu++17 -MMD -isystem "{runtime.tools.riscv-gcc.path}/riscv-nuclei-elf/include/c++/9.2.0/riscv-nuclei-elf/rv32imafc/ilp32f" -fno-exceptions -fno-rtti -fno-use-cxa-atexit -fno-threadsafe-statics
compiler.S.flags={compiler.common.flags} -g -x assembler-with-cpp -MMD
compiler.ar.flags=crs

compiler.c.extra_flags=
compiler.cpp.extra_flags=
compiler.S.extra_flags=
# Binary-only vendor archives retain stale GCC 9.2 LTO type metadata. Suppress
# only that known final-link diagnostic; source compilation warnings stay on.
compiler.c.elf.extra_flags=-Wno-lto-type-mismatch
compiler.ar.extra_flags=
compiler.objcopy.extra_flags=
compiler.libraries.ldflags=

compiler.c.elf.flags={compiler.common.flags} -g -nostartfiles --specs=nano.specs -Wl,--gc-sections -Wl,-u,_start -Wl,-u,main "-Wl,--Map={build.path}/{build.project_name}.map" "-L{compiler.sdk.library.path}/{build.variant}" "-L{compiler.sdk.library.path}" "-L{runtime.platform.path}/tools/sdk/ld" "-T{runtime.platform.path}/tools/sdk/ld/ci130x_alg_pro_{build.algorithm}.lds"
compiler.c.elf.wrap=-Wl,--wrap,memcmp -Wl,--wrap,memcpy -Wl,--wrap,memmove -Wl,--wrap,memset -Wl,--wrap,strcat -Wl,--wrap,strcmp -Wl,--wrap,strcpy -Wl,--wrap,strlen -Wl,--wrap,strncat -Wl,--wrap,strncmp -Wl,--wrap,strncpy -Wl,--wrap,strstr -Wl,--wrap,stpcpy -Wl,--wrap,stpncpy -Wl,--wrap,strnlen -Wl,--wrap,strtoul -Wl,--wrap,atol -Wl,--wrap,qsort -Wl,--wrap,malloc -Wl,--wrap,free -Wl,--wrap,vsnprintf -Wl,--wrap,sprintf -Wl,--wrap,sscanf -Wl,--wrap,abs -Wl,--wrap,isnan -Wl,--wrap,isinf -Wl,--wrap,cosf -Wl,--wrap,sinf -Wl,--wrap,tanf -Wl,--wrap,expf -Wl,--wrap,sqrtf -Wl,--wrap,fabsf -Wl,--wrap,logf -Wl,--wrap,log10f -Wl,--wrap,atoi -Wl,--wrap,vTaskStartScheduler -Wl,--wrap,deal_asr_msg_by_cmd_id -Wl,--wrap,sys_asr_result_hook -Wl,--wrap,ADC_IRQHandler
compiler.c.elf.libs=-lasr_dis_deepse_v2 -lir_data -lnlp -lnewlib_port -lfreertos_port -ldsu -lflash_encrypt -lcwsl_v2 -ltts -lOnMicroBLE -lcias_crypto -lcikd_pro_cwsl_dis_frm -lm

# Windows PowerShell hooks
recipe.hooks.prebuild.1.pattern.windows=powershell.exe -NoProfile -ExecutionPolicy Bypass -File "{runtime.platform.path}/tools/prepare_resources.ps1" -ProjectPath "{build.source.path}" -PackageResources "{runtime.platform.path}/recursos" -Algorithm "{build.algorithm}"
recipe.hooks.prebuild.2.pattern.windows=powershell.exe -NoProfile -ExecutionPolicy Bypass -File "{runtime.platform.path}/tools/compile_sdk_sources.ps1" -PlatformPath "{runtime.platform.path}" -BuildPath "{build.path}" -Compiler "{compiler.path}{compiler.c.cmd}" -VariantPath "{build.variant.path}" -BuildExtraFlags "{build.extra_flags} {build.clock_flags}"

# macOS/Linux Python hooks
recipe.hooks.prebuild.1.pattern.linux=python3 "{runtime.platform.path}/tools/prepare_resources.py" --project-path "{build.source.path}" --package-resources "{runtime.platform.path}/recursos" --algorithm "{build.algorithm}"
recipe.hooks.prebuild.2.pattern.linux=python3 "{runtime.platform.path}/tools/compile_sdk_sources.py" --platform-path "{runtime.platform.path}" --build-path "{build.path}" --compiler "{compiler.path}{compiler.c.cmd}" --variant-path "{build.variant.path}" --build-extra-flags "{build.extra_flags} {build.clock_flags}"
recipe.hooks.prebuild.1.pattern.macosx=python3 "{runtime.platform.path}/tools/prepare_resources.py" --project-path "{build.source.path}" --package-resources "{runtime.platform.path}/recursos" --algorithm "{build.algorithm}"
recipe.hooks.prebuild.2.pattern.macosx=python3 "{runtime.platform.path}/tools/compile_sdk_sources.py" --platform-path "{runtime.platform.path}" --build-path "{build.path}" --compiler "{compiler.path}{compiler.c.cmd}" --variant-path "{build.variant.path}" --build-extra-flags "{build.extra_flags} {build.clock_flags}"

recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} {compiler.c.extra_flags} -DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_CI13XX {compiler.sdk.defines} {build.extra_flags} {build.clock_flags} {compiler.sdk.includes} {includes} -c "{source_file}" -o "{object_file}"
recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} {compiler.cpp.extra_flags} -DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_CI13XX {compiler.sdk.defines} {build.extra_flags} {build.clock_flags} {compiler.sdk.includes} {includes} -c "{source_file}" -o "{object_file}"
recipe.S.o.pattern="{compiler.path}{compiler.S.cmd}" {compiler.S.flags} {compiler.S.extra_flags} -DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_CI13XX {compiler.sdk.defines} {build.extra_flags} {build.clock_flags} {compiler.sdk.includes} {includes} -c "{source_file}" -o "{object_file}"
recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" {compiler.ar.flags} {compiler.ar.extra_flags} "{archive_file_path}" "{object_file}"

# The source-available vendor startup, SDK main(), FreeRTOS kernel, drivers and
# application glue are compiled by the prebuild hook. The response file links
# those objects directly; only components distributed without source remain as
# vendor archives. The Arduino core wraps scheduler startup to add setup()/loop().
recipe.c.combine.pattern="{compiler.path}{compiler.c.elf.cmd}" {compiler.c.elf.flags} {compiler.c.elf.wrap} {compiler.c.elf.extra_flags} -o "{build.path}/{build.project_name}.elf" -Wl,--start-group "@{build.path}/ci13xx_sdk_objects.rsp" {object_files} "{archive_file_path}" {compiler.libraries.ldflags} {compiler.c.elf.libs} -Wl,--end-group -Wl,-EL

# First build the CI13XX two-image user-code container. If the sketch declares
# COMMAND/VOICE resource macros, generate fresh resources through ci-service;
# then compose and inspect the complete image uploaded from Flash address 0.
# Windows PowerShell postbuild
recipe.objcopy.bin.pattern.windows=powershell.exe -NoProfile -ExecutionPolicy Bypass -File "{runtime.platform.path}/tools/postbuild.ps1" -Elf "{build.path}/{build.project_name}.elf" -Source "{build.source.path}/{build.project_name}" -Output "{build.path}/{build.project_name}.user_code.bin" -FirmwareOutput "{build.path}/{build.project_name}.bin" -Objcopy "{compiler.path}{compiler.objcopy.cmd}" -PlatformPath "{runtime.platform.path}" -CitoolCli "{runtime.tools.citool-cli.path}/{tools.citool-cli.cmd}" -ServiceUrl "{build.ci_service_url}" -ProjectResources "{build.source.path}/recursos" -Chip "{build.chip}" -Algorithm "{build.algorithm}" -MaxUserCodeSize {build.user_code_max}

# macOS/Linux Python postbuild
recipe.objcopy.bin.pattern.linux=python3 "{runtime.platform.path}/tools/postbuild.py" --elf "{build.path}/{build.project_name}.elf" --source "{build.source.path}/{build.project_name}" --output "{build.path}/{build.project_name}.user_code.bin" --firmware-output "{build.path}/{build.project_name}.bin" --objcopy "{compiler.path}{compiler.objcopy.cmd}" --platform-path "{runtime.platform.path}" --citool-cli "{runtime.tools.citool-cli.path}/{tools.citool-cli.cmd}" --service-url "{build.ci_service_url}" --project-resources "{build.source.path}/recursos" --chip "{build.chip}" --algorithm "{build.algorithm}" --max-user-code-size {build.user_code_max}
recipe.objcopy.bin.pattern.macosx=python3 "{runtime.platform.path}/tools/postbuild.py" --elf "{build.path}/{build.project_name}.elf" --source "{build.source.path}/{build.project_name}" --output "{build.path}/{build.project_name}.user_code.bin" --firmware-output "{build.path}/{build.project_name}.bin" --objcopy "{compiler.path}{compiler.objcopy.cmd}" --platform-path "{runtime.platform.path}" --citool-cli "{runtime.tools.citool-cli.path}/{tools.citool-cli.cmd}" --service-url "{build.ci_service_url}" --project-resources "{build.source.path}/recursos" --chip "{build.chip}" --algorithm "{build.algorithm}" --max-user-code-size {build.user_code_max}

recipe.output.tmp_file={build.project_name}.bin
recipe.output.save_file={build.project_name}.{build.variant}.firmware.bin

recipe.size.pattern="{compiler.path}{compiler.size.cmd}" -A "{build.path}/{build.project_name}.elf"
recipe.size.regex=^(?:\.boot_parameter|\.init|\.ilalign|\.text|\.ro_data|\.data|\.preinit_array|\.init_array|\.fini_array|\.ctors|\.dtors|\.gcc_except_table\S*)\s+([0-9]+).*
recipe.size.regex.data=^(?:\.data|\.bss|\.no_init|\.stack|\.sys_heap)\s+([0-9]+).*
